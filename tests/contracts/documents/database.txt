19. 收尾：updated_at 自动维护触发器
-- 通用函数：设置 updated_at = now()CREATE OR REPLACE FUNCTION llm_set_updated_at() RETURNS TRIGGER AS $$BEGIN  NEW.updated_at = now();  RETURN NEW;END;$$ LANGUAGE plpgsql;-- 为所有含 updated_at 的表批量创建 BEFORE UPDATE 触发器DO $$DECLARE  t TEXT;BEGIN  FOREACH t IN ARRAY ARRAY[    'llm_knowledge_base', 'llm_document', 'llm_parsed_document',    'llm_node_parent', 'llm_node_child', 'llm_node_summary', 'llm_question',    'llm_doc_trace',    'llm_review_record', 'llm_integration_app'  ]  LOOP    EXECUTE format('DROP TRIGGER IF EXISTS trg_%s_set_updated_at ON %I;', t, t);    EXECUTE format(      'CREATE TRIGGER trg_%s_set_updated_at BEFORE UPDATE ON %I       FOR EACH ROW EXECUTE FUNCTION llm_set_updated_at();',      t, t);  END LOOP;END;$$;
